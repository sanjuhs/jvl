01-loan-vs-investment.lvl
fact transfer_17 : TransferOfValue {
    from: A  to: B  amount: INR 1_000_000  on: 2025-03-10
} from source(doc="BankStatement_3", page=2, para=4) status Established

claim c_loan by A : "the transfer was a loan" asserts Loan(transfer_17)

evidence w17 : Message { author: B  text: "I'll repay next month." }
    from source(doc="WhatsApp_17", para=21) supports RepaymentObligation(transfer_17)

rule loan_definition:
    Loan(t) requires TransferOfValue(t) RepaymentObligation(t)

assert Loan(transfer_17) under BalanceOfProbabilities
$ lvl assert 01-loan-vs-investment.lvl
⚖  assert  Loan(transfer_17)   standard: BalanceOfProbabilities

  Loan(transfer_17) ............................. SUPPORTED
    └─ requires: TransferOfValue(transfer_17) ∧ RepaymentObligation(transfer_17)
      TransferOfValue(transfer_17) .............. ESTABLISHED
        └─ Established fact (transfer_17)  →  BankStatement_3 p.2 ¶4
      RepaymentObligation(transfer_17) .......... SUPPORTED
        └─ supporting evidence (w17)  →  WhatsApp_17 ¶21

  RESULT: SUPPORTED  ✓ (meets BalanceOfProbabilities)
  NOTE:   contested by claim c_invest (B: Investment(transfer_17))
03-cheating-s420.lvl
rule cheating:
    Cheating(p) requires
        Deception(p)
        DishonestInducement(p)
        DeliveryOfProperty(p)
        DishonestIntentionAtInception(p)

evidence brochure : Document { note: "fabricated projection" }
    from source(doc="Exhibit_P3", para=8) supports Deception(payment)

# No evidence establishes dishonest intention at inception.

assert Cheating(payment) under BeyondReasonableDoubt
$ lvl discover 03-cheating-s420.lvl "Cheating(payment)"
⚖  discover  what is missing for Cheating(payment)

  ✗ DishonestIntentionAtInception(payment) — UNKNOWN

# The offence is a conjunction of elements. Three are supported,
# but the mental element is unproven — so under the criminal
# standard the charge cannot be certified. The compiler says so,
# and points at exactly the gap.
02-nda-contract.lvl
fact nda : Agreement {
    penalty_cap: INR 5_000_000  expires_on: 2028-01-15
} from source(doc="NDA_executed", page=1) status Established

fact disclosure_event : Disclosure {
    on: 2025-06-20  amount: INR 8_000_000
} from source(doc="Complaint", page=3, para=12) status Alleged

constraint damages_within_cap: disclosure_event.amount <= nda.penalty_cap
constraint disclosure_before_expiry: disclosure_event.on before nda.expires_on
$ lvl constraints 02-nda-contract.lvl
⚖  objective constraints: 02-nda-contract.lvl

  ✗ damages_within_cap: VIOLATED
  ✓ disclosure_after_signing: holds
  ✓ disclosure_before_expiry: holds

# Claimed damages (INR 8,000,000) exceed the contractual cap
# (INR 5,000,000). Pure arithmetic — no legal judgement needed,
# and the kind of defect a prose review routinely misses.
v1 vs v2
# v1
fact agreement : ServiceAgreement {
    fee: INR 500_000  due_on: 2025-09-30
} from source(doc="MSA_v1", page=1) status Established

# v2 — fee raised, penalty clause added
fact agreement : ServiceAgreement {
    fee: INR 650_000  due_on: 2025-09-30  penalty: INR 50_000
} from source(doc="MSA_v2", page=1) status Established

rule penalty_clause:
    PenaltyOwed(a) requires ServiceAgreement(a) LateDelivery(a)
$ lvl equiv v1.lvl v2.lvl
⚖  equiv  service-agreement-v1  ≟  service-agreement-v2

  ✗ DIFFERENT — they diverge on these propositions:

    · proposition only in second — PenaltyOwed(agreement) (UNKNOWN)
    · term changed — agreement.fee: INR500000 → INR650000
    · term changed — agreement.penalty: None → INR50000
    · rule added — penalty_clause

# Ordering, comments and formatting never register — only a real
# change in meaning does. "Do these two contracts say the same
# thing?" becomes a command.
